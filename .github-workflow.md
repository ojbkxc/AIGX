# AIGX GitHub Actions 配置文件
# 复制以下内容为 .github/workflows/github-actions.yml

name: AIGX Multi-platform Build

on:
  push:
    branches: [ main, develop ]
  release:
    types: [ created ]
  workflow_dispatch:

env:
  RUST_TOOLCHAIN: stable

jobs:
  # Linux AMD64 构建
  build-linux-amd64:
    name: Build Linux AMD64
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Set up Rust
        uses: actions-rs/toolchain@v1
        with:
          profile: minimal
          toolchain: ${{ env.RUST_TOOLCHAIN }}
          override: true

      - name: Build Linux AMD64
        run: |
          cargo build --release
          mkdir -p build/2024-Linux-x86_64
          mv target/release/aigx build/2024-Linux-x86_64/

      - uses: actions/upload-artifact@v4
        with:
          name: aigx-linux-amd64
          path: build/2024-Linux-x86_64/

  # Linux ARM64 构建
  build-linux-arm64:
    name: Build Linux ARM64
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Set up Rust
        uses: actions-rs/toolchain@v1
        with:
          profile: minimal
          toolchain: ${{ env.RUST_TOOLCHAIN }}
          override: true

      - name: Install cross compiler for ARM64
        run: sudo apt-get update && sudo apt-get install -y gcc-aarch64-linux-gnu

      - name: Build Linux ARM64
        run: |
          cargo build --release --target aarch64-unknown-linux-gnu
          mkdir -p build/2024-Linux-arm64
          mv target/aarch64-unknown-linux-gnu/release/aigx build/2024-Linux-arm64/

      - uses: actions/upload-artifact@v4
        with:
          name: aigx-linux-arm64
          path: build/2024-Linux-arm64/

  # Windows AMD64 构建
  build-windows-amd64:
    name: Build Windows AMD64
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4

      - name: Set up Rust
        uses: actions-rs/toolchain@v1
        with:
          profile: minimal
          toolchain: ${{ env.RUST_TOOLCHAIN }}
          override: true

      - name: Build Windows AMD64
        run: |
          cargo build --release
          mkdir -p build/2024-Windows-x86_64
          copy target\release\aigx.exe build\2024-Windows-x86_64\

      - uses: actions/upload-artifact@v4
        with:
          name: aigx-windows-amd64
          path: build/2024-Windows-x86_64/

  # 创建 Release 测试
  create-test-release:
    name: Create Test Release
    needs: [build-linux-amd64, build-linux-arm64, build-windows-amd64]
    runs-on: ubuntu-latest
    if: github.event_name == 'workflow_dispatch'

    steps:
      - uses: actions/download-artifact@v4
        with:
          path: artifacts

      - name: Upload to GitHub Releases
        uses: softprops/action-gh-release@v1
        with:
          tag_name: test-${{ github.run_number }}
          name: Test Build ${{ github.run_number }}
          files: artifacts/**
          generate_release_notes: true
